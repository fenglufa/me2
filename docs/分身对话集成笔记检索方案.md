# 分身对话集成笔记检索方案

## 一、需求背景

用户希望在与分身对话时，分身能够访问和理解用户的笔记内容，实现真正的"记忆"功能。

### 使用场景

1. **查询消费记录**
   - 用户："昨天我吃了什么，花了多少钱？"
   - 分身："你昨天吃了蛋炒饭，花了 10 块钱哦~"

2. **查询笔记内容**
   - 用户："我的笔记中有哪些是关于 golang 开发的？"
   - 分身："你有 3 条关于 golang 的笔记：1. golang 并发编程学习... 2. ..."

3. **查询 TODO**
   - 用户："我今天有什么任务要做？"
   - 分身："你今天要买牛奶和完成项目报告哦"

4. **查询情绪记录**
   - 用户："我最近心情怎么样？"
   - 分身："最近你的情绪有些波动，上周三你说工作压力大..."

## 二、技术方案：RAG（检索增强生成）

### 核心思路

在对话时，根据用户问题检索相关笔记，将笔记内容作为上下文传递给 AI，让分身基于笔记内容回答问题。

### 实现流程

```
用户问："昨天我吃了什么，花了多少钱？"
    ↓
1. Dialogue Service 接收问题
    ↓
2. 调用 Note Service 检索相关笔记
   - 关键词：昨天、吃、花费
   - 时间范围：昨天
   - 类型：expense
    ↓
3. 获取相关笔记：
   "昨天吃了蛋炒饭，花费 10 块钱"
    ↓
4. 构建上下文 Prompt：
   系统提示词 + 分身信息 + 最近事件 + 【相关笔记】 + 对话历史
    ↓
5. 调用 AI Service 生成回复
    ↓
6. 分身回复："你昨天吃了蛋炒饭，花了 10 块钱哦~"
```

**方式 A：简单关键词匹配（MVP - Phase 1）**

```go
// search_notes_logic.go
package logic

import (
    "context"
    "time"

    "github.com/me2/note/rpc/internal/svc"
    "github.com/me2/note/rpc/note"
    "github.com/zeromicro/go-zero/core/logx"
)

type SearchNotesLogic struct {
    ctx    context.Context
    svcCtx *svc.ServiceContext
    logx.Logger
}

func NewSearchNotesLogic(ctx context.Context, svcCtx *svc.ServiceContext) *SearchNotesLogic {
    return &SearchNotesLogic{
        ctx:    ctx,
        svcCtx: svcCtx,
        Logger: logx.WithContext(ctx),
    }
}

func (l *SearchNotesLogic) SearchNotes(in *note.SearchNotesRequest) (*note.SearchNotesResponse, error) {
    // 1. 解析时间范围
    startTime, endTime := parseTimeRange(in.DateRange)

    // 2. 构建 SQL 查询（使用 LIKE 进行关键词匹配）
    notes, err := l.svcCtx.NoteModel.Search(
        in.UserId,
        in.Query,      // 关键词
        in.Types,      // 类型过滤
        startTime,
        endTime,
        in.Limit,
    )
    if err != nil {
        l.Errorf("搜索笔记失败: %v", err)
        return nil, err
    }

    // 3. 转换为响应格式
    var noteInfos []*note.NoteInfo
    for _, n := range notes {
        noteInfos = append(noteInfos, &note.NoteInfo{
            Id:        n.Id,
            RawText:   n.RawText,
            AiSummary: n.AiSummary,
            Types:     parseTypes(n.Types),
            CreatedAt: n.CreatedAt.Format("2006-01-02 15:04:05"),
        })
    }

    return &note.SearchNotesResponse{Notes: noteInfos}, nil
}

// 解析时间范围
func parseTimeRange(dateRange string) (time.Time, time.Time) {
    now := time.Now()
    var startTime, endTime time.Time

    switch dateRange {
    case "today":
        startTime = time.Date(now.Year(), now.Month(), now.Day(), 0, 0, 0, 0, now.Location())
        endTime = now
    case "yesterday":
        yesterday := now.AddDate(0, 0, -1)
        startTime = time.Date(yesterday.Year(), yesterday.Month(), yesterday.Day(), 0, 0, 0, 0, now.Location())
        endTime = time.Date(yesterday.Year(), yesterday.Month(), yesterday.Day(), 23, 59, 59, 0, now.Location())
    case "last_week":
        startTime = now.AddDate(0, 0, -7)
        endTime = now
    case "last_month":
        startTime = now.AddDate(0, -1, 0)
        endTime = now
    default:
        // 默认查询最近 7 天
        startTime = now.AddDate(0, 0, -7)
        endTime = now
    }

    return startTime, endTime
}
```

```go
// note_model.go
func (m *defaultNoteModel) Search(userId int64, query string, types []string, startTime, endTime time.Time, limit int32) ([]*Note, error) {
    var notes []*Note

    // 构建查询
    sql := `SELECT id, user_id, avatar_id, raw_text, ai_summary, types, emotion_data, created_at, updated_at
            FROM notes
            WHERE user_id = ?
              AND raw_text LIKE ?
              AND created_at BETWEEN ? AND ?`

    args := []interface{}{userId, "%" + query + "%", startTime, endTime}

    // 如果指定了类型，添加类型过滤
    if len(types) > 0 {
        // JSON_CONTAINS 或 JSON_OVERLAPS 查询
        sql += " AND JSON_OVERLAPS(types, ?)"
        typesJson, _ := json.Marshal(types)
        args = append(args, string(typesJson))
    }

    sql += " ORDER BY created_at DESC LIMIT ?"
    args = append(args, limit)

    err := m.conn.QueryRows(&notes, sql, args...)
    return notes, err
}
```

```go
// 构建系统 Prompt（新增 userQuery 参数）
func (b *ContextBuilder) BuildSystemPrompt(avatarID, sessionID int64, userQuery string) (string, error) {
    // 1. 获取分身信息
    avatarResp, err := b.svcCtx.AvatarRpc.GetAvatarInfo(b.ctx, &avatar_client.GetAvatarInfoRequest{
        AvatarId: avatarID,
    })
    if err != nil {
        return "", err
    }

    avatarInfo := b.formatAvatarInfo(avatarResp.Avatar)

    // 2. 获取最近事件
    recentEvents, err := b.getRecentEvents(avatarID)
    if err != nil {
        return "", err
    }

    // 3. 获取对话历史
    history, err := b.getConversationHistory(sessionID)
    if err != nil {
        return "", err
    }

    // 4. 【新增】获取相关笔记
    relevantNotes, err := b.getRelevantNotes(avatarResp.Avatar.UserId, userQuery)
    if err != nil {
        // 笔记检索失败不影响对话，使用默认值
        relevantNotes = "暂无相关笔记"
    }

    // 5. 构建完整 Prompt
    prompt := fmt.Sprintf(`你是用户的AI分身,具有以下基本信息:
%s

最近发生的事件:
%s

用户的相关笔记:
%s

对话历史:
%s

请以分身的口吻回复用户。回复要自然、简洁,不要过于正式。
如果用户问到笔记中的内容，请基于"用户的相关笔记"部分进行回答。`,
        avatarInfo, recentEvents, relevantNotes, history)

    return prompt, nil
}

// 【新增方法】获取相关笔记
func (b *ContextBuilder) getRelevantNotes(userID int64, query string) (string, error) {
    // 如果查询为空，不检索笔记
    if query == "" {
        return "暂无相关笔记", nil
    }

    // 调用 Note Service 检索
    notesResp, err := b.svcCtx.NoteRpc.SearchNotes(b.ctx, &note_client.SearchNotesRequest{
        UserId: userID,
        Query:  query,
        Limit:  5, // 最多返回 5 条相关笔记
    })

    if err != nil || len(notesResp.Notes) == 0 {
        return "暂无相关笔记", nil
    }

    // 格式化笔记
    var notes []string
    for i, n := range notesResp.Notes {
        notes = append(notes, fmt.Sprintf("%d. [%s] %s",
            i+1,
            formatTime(n.CreatedAt),
            n.RawText))
    }

    return strings.Join(notes, "\n"), nil
}
```

### 5.2 缓存优化

对于频繁查询的笔记，可以使用 Redis 缓存：

```go
// 缓存 key: note:search:{user_id}:{query_hash}
func (l *SearchNotesLogic) SearchNotes(in *note.SearchNotesRequest) (*note.SearchNotesResponse, error) {
    // 1. 尝试从缓存获取
    cacheKey := fmt.Sprintf("note:search:%d:%s", in.UserId, hashQuery(in.Query))
    cached, err := l.svcCtx.Redis.Get(cacheKey)
    if err == nil && cached != "" {
        var resp note.SearchNotesResponse
        json.Unmarshal([]byte(cached), &resp)
        return &resp, nil
    }

    // 2. 缓存未命中，查询数据库
    notes, err := l.svcCtx.NoteModel.Search(...)

    // 3. 写入缓存（5 分钟过期）
    respJson, _ := json.Marshal(resp)
    l.svcCtx.Redis.Setex(cacheKey, string(respJson), 300)

    return resp, nil
}
```

## 八、注意事项

### 8.1 隐私和安全

- 笔记检索必须验证 user_id，防止越权访问
- 敏感信息（如密码、身份证号）不应出现在对话上下文中
- 考虑添加笔记隐私级别（公开/私密）

### 8.2 性能考虑

- 限制每次检索返回的笔记数量（默认 5 条）
- 使用索引优化查询性能
- 考虑异步检索（不阻塞对话）
- 添加缓存减少数据库压力

### 8.3 用户体验

- 检索失败不应影响对话（降级处理）
- 分身回答要自然，不要生硬地列举笔记
- 支持模糊匹配，提高容错性
- 时间范围解析要准确（昨天、上周、最近）
